# Dependencies for the price-estimation web service only (kz/web).
#
# Collection requires Playwright and a complete browser; inference does not.
# Keeping a separate file avoids downloading hundreds of unused megabytes for
# every web deployment.
#
# Keep these versions aligned with requirements.txt. A model trained against a
# different runtime stack might not load correctly in the container.

fastapi==0.141.1
# Multipart form parsing for seller photo uploads (/api/photos/check).
# FastAPI raises at import of request.form() without it.
python-multipart==0.0.32

# Pillow alone, deliberately without imagehash. Pillow costs 13 MB and buys
# the readable-image check, pixel dimensions, and the "shows bodywork" axis
# whenever the CLIP stack is present. imagehash would add scipy and
# PyWavelets, roughly 80 MB against a 512 MB free-instance ceiling, to buy
# only perceptual near-duplicates; exact duplicates are already detected by
# content hash with no dependency at all. kz/web/photo_intake degrades to
# exact-only and says so.
Pillow==12.3.0
uvicorn[standard]==0.52.1

# Model and preprocessing
catboost==1.2.10
pandas==2.3.3
numpy==2.5.1
scikit-learn==1.9.0

# The database only decorates results with comparable cars. Core inference
# remains available without it; see kz/web/service.py:query.
SQLAlchemy==2.0.51
psycopg2-binary==2.9.12
python-dotenv==1.2.2
